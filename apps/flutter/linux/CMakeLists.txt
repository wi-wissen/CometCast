# Compilation settings that should be applied to most targets.
#
# Be cautious about adding new options here, as plugins use this function by
# default. In most cases, you should add new options to specific targets instead
# of modifying this function.
function(APPLY_STANDARD_SETTINGS TARGET)
  target_compile_features(${TARGET} PUBLIC cxx_std_14)
  target_compile_options(${TARGET} PRIVATE -Wall -Werror)
  target_compile_options(${TARGET} PRIVATE "$<$<NOT:$<CONFIG:Debug>>:-O3>")
  target_compile_definitions(${TARGET} PRIVATE "$<$<NOT:$<CONFIG:Debug>>:NDEBUG>")
endfunction()

# --- Comet Cast: build the Rust core (crates/cc-ffi) and bundle it ---------------------
# Not generated by Flutter. This runs `cargo build -p cc-ffi` as part of the normal
# `flutter build linux` / `flutter run -d linux` and installs libcc_ffi.so next to the
# Flutter library (bundle/lib/), where the Dart code looks for it.
# Debug builds use the cargo "dev" profile, Profile/Release use `--release`.
# See docs/flutter-bridge.md.
get_filename_component(CC_REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
find_program(CARGO_EXECUTABLE cargo HINTS "$ENV{HOME}/.cargo/bin" "$ENV{CARGO_HOME}/bin")
if(NOT CARGO_EXECUTABLE)
  message(FATAL_ERROR "cargo not found: install Rust (https://rustup.rs) to build the Comet Cast core.")
endif()
if(CMAKE_BUILD_TYPE MATCHES "Debug")
  set(CC_FFI_PROFILE "debug")
  set(CC_FFI_CARGO_FLAGS "")
else()
  set(CC_FFI_PROFILE "release")
  set(CC_FFI_CARGO_FLAGS "--release")
endif()
# Own target dir inside the Flutter build folder, so the result does not depend on the
# CARGO_TARGET_DIR variable of the person building.
set(CC_FFI_TARGET_DIR "${CMAKE_BINARY_DIR}/cargo-target")
# Optional: COMETCAST_CARGO_TARGET_DIR=/some/dir shares one cargo target folder with other
# builds (saves disk and time; used for the checks in apps/flutter/README.md).
if(DEFINED ENV{COMETCAST_CARGO_TARGET_DIR} AND NOT "$ENV{COMETCAST_CARGO_TARGET_DIR}" STREQUAL "")
  set(CC_FFI_TARGET_DIR "$ENV{COMETCAST_CARGO_TARGET_DIR}")
endif()
set(CC_FFI_LIB "${CC_FFI_TARGET_DIR}/${CC_FFI_PROFILE}/libcc_ffi.so")
# The PDF render worker (crates/cc-docs, program `cc-docs-worker`): a separate, resource-limited
# child process that renders received PDFs (docs/documents.md). `-p cc-docs` builds its binary
# (feature `worker-bin`, on by default). It is installed next to the executable, where the Rust
# core looks for it (crates/cc-ffi/src/content.rs, `worker()`).
set(CC_DOCS_WORKER "${CC_FFI_TARGET_DIR}/${CC_FFI_PROFILE}/cc-docs-worker")
# The Office conversion worker (crates/cc-office, program `cc-office-worker`): converts .docx,
# .pptx and .xlsx to PDF on the sender (docs/documents.md). `-p cc-office` builds its binary
# (feature `worker-bin`, on by default) with the pure-Rust engine (feature `engine-rust`, also
# default). Installed next to the executable like the PDF worker; `cc-ffi` finds it there
# (crates/cc-ffi/src/office.rs). It is built by its own cargo command: `cc-ffi` depends on
# `cc-office` without the engine, and building both in one command would unify the features
# and link the engine (Typst) into libcc_ffi.so again.
set(CC_OFFICE_WORKER "${CC_FFI_TARGET_DIR}/${CC_FFI_PROFILE}/cc-office-worker")
# The Wayland screen capture (desktop portal + PipeWire, crates/cc-capture-pw) is a separate
# library, `libcc_capture_pw.so`, that the app loads with dlopen when the person shares the screen
# (decision N57). libcc_ffi.so does not link libpipewire, so the app starts on a machine without
# it and then reports `capture_no_portal`. It is built when the PipeWire development files are
# there (`pkg-config libpipewire-0.3`, libclang for the bindings), in its own cargo command
# because of its feature `plugin`. Without them the app still builds; Wayland sharing then
# reports `capture_no_portal` and X11 sessions capture as before. Packages: depend on
# `libpipewire-0.3-0` (deb) / `pipewire-libs` (rpm) as a *recommended* package, not a hard one.
find_package(PkgConfig QUIET)
set(CC_CAPTURE_PW_LIB "${CC_FFI_TARGET_DIR}/${CC_FFI_PROFILE}/libcc_capture_pw.so")
set(CC_CAPTURE_PW_BUILD FALSE)
if(PkgConfig_FOUND)
  pkg_check_modules(CC_PIPEWIRE QUIET libpipewire-0.3)
  if(CC_PIPEWIRE_FOUND)
    set(CC_CAPTURE_PW_BUILD TRUE)
  endif()
endif()
if(NOT CC_CAPTURE_PW_BUILD)
  message(STATUS "libpipewire-0.3 development files not found: Wayland screen capture is not built (capture_no_portal)")
endif()
add_custom_target(cc_ffi_build ALL
  COMMAND ${CARGO_EXECUTABLE} build -p cc-ffi -p cc-docs ${CC_FFI_CARGO_FLAGS}
          --manifest-path "${CC_REPO_ROOT}/Cargo.toml"
          --target-dir "${CC_FFI_TARGET_DIR}"
  COMMAND ${CARGO_EXECUTABLE} build -p cc-office ${CC_FFI_CARGO_FLAGS}
          --manifest-path "${CC_REPO_ROOT}/Cargo.toml"
          --target-dir "${CC_FFI_TARGET_DIR}"
  BYPRODUCTS "${CC_FFI_LIB}" "${CC_DOCS_WORKER}" "${CC_OFFICE_WORKER}"
  COMMENT "Building the Rust core (cc-ffi), the PDF worker (cc-docs-worker) and the Office worker (cc-office-worker, ${CC_FFI_PROFILE})"
  USES_TERMINAL)
if(CC_CAPTURE_PW_BUILD)
  add_custom_target(cc_capture_pw_build ALL
    COMMAND ${CARGO_EXECUTABLE} build -p cc-capture-pw --features plugin ${CC_FFI_CARGO_FLAGS}
            --manifest-path "${CC_REPO_ROOT}/Cargo.toml"
            --target-dir "${CC_FFI_TARGET_DIR}"
    BYPRODUCTS "${CC_CAPTURE_PW_LIB}"
    COMMENT "Building the Wayland capture library (cc-capture-pw, ${CC_FFI_PROFILE})"
    USES_TERMINAL)
  add_dependencies(cc_capture_pw_build cc_ffi_build)
  add_dependencies(${BINARY_NAME} cc_capture_pw_build)
endif()
add_dependencies(${BINARY_NAME} cc_ffi_build)

# Only the install-generated bundle's copy of the executable will launch
# correctly, since the resources must in the right relative locations. To avoid
# people trying to run the unbundled copy, put it in a subdirectory instead of
# the default top-level location.
set_target_properties(${BINARY_NAME}
  PROPERTIES
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/intermediates_do_not_run"
)

# === Installation ===
# By default, "installing" just makes a relocatable bundle in the build
# directory.
set(BUILD_BUNDLE_DIR "${PROJECT_BINARY_DIR}/bundle")
if(CMAKE_INSTALL_PREFIX_INITIALIZED_TO_DEFAULT)
  set(CMAKE_INSTALL_PREFIX "${BUILD_BUNDLE_DIR}" CACHE PATH "..." FORCE)
endif()

# Start with a clean build bundle directory every time.
install(CODE "
  file(REMOVE_RECURSE \"${BUILD_BUNDLE_DIR}/\")
  " COMPONENT Runtime)

set(INSTALL_BUNDLE_DATA_DIR "${CMAKE_INSTALL_PREFIX}/data")
set(INSTALL_BUNDLE_LIB_DIR "${CMAKE_INSTALL_PREFIX}/lib")

install(TARGETS ${BINARY_NAME} RUNTIME DESTINATION "${CMAKE_INSTALL_PREFIX}"
  COMPONENT Runtime)

install(FILES "${FLUTTER_ICU_DATA_FILE}" DESTINATION "${INSTALL_BUNDLE_DATA_DIR}"
  COMPONENT Runtime)

install(FILES "${FLUTTER_LIBRARY}" DESTINATION "${INSTALL_BUNDLE_LIB_DIR}"
  COMPONENT Runtime)

# The Rust core (see the cc_ffi_build target above).
install(FILES "${CC_FFI_LIB}" DESTINATION "${INSTALL_BUNDLE_LIB_DIR}"
  COMPONENT Runtime)

# The Wayland capture library, next to libcc_ffi.so (bundle/lib/); the Rust core loads it from
# `lib/` beside the executable (cc_capture::pwplugin). Optional: see the target above.
if(CC_CAPTURE_PW_BUILD)
  install(FILES "${CC_CAPTURE_PW_LIB}" DESTINATION "${INSTALL_BUNDLE_LIB_DIR}"
    COMPONENT Runtime)
endif()

# The desktop entry of the application id: the portal's screen picker names "Comet Cast" from
# it (decision N57; the core registers the id `${APPLICATION_ID}` with the portal). A package
# installs it as /usr/share/applications/${APPLICATION_ID}.desktop.
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/${APPLICATION_ID}.desktop"
  DESTINATION "${CMAKE_INSTALL_PREFIX}/share/applications"
  COMPONENT Runtime)

# The PDF render worker, a program next to the executable (bundle/cc-docs-worker). Not in lib/:
# cc-docs and the Rust core look for it in the directory of the running executable.
install(PROGRAMS "${CC_DOCS_WORKER}" DESTINATION "${CMAKE_INSTALL_PREFIX}"
  COMPONENT Runtime)

# The Office conversion worker, next to the executable too (bundle/cc-office-worker).
install(PROGRAMS "${CC_OFFICE_WORKER}" DESTINATION "${CMAKE_INSTALL_PREFIX}"
  COMPONENT Runtime)

foreach(bundled_library ${PLUGIN_BUNDLED_LIBRARIES})
  install(FILES "${bundled_library}"
    DESTINATION "${INSTALL_BUNDLE_LIB_DIR}"
    COMPONENT Runtime)
endforeach(bundled_library)

# Copy the native assets provided by the build.dart from all packages.
set(NATIVE_ASSETS_DIR "${PROJECT_BUILD_DIR}native_assets/linux/")
install(DIRECTORY "${NATIVE_ASSETS_DIR}"
   DESTINATION "${INSTALL_BUNDLE_LIB_DIR}"
   COMPONENT Runtime)

# Fully re-copy the assets directory on each build to avoid having stale files
# from a previous install.
set(FLUTTER_ASSET_DIR_NAME "flutter_assets")
install(CODE "
  file(REMOVE_RECURSE \"${INSTALL_BUNDLE_DATA_DIR}/${FLUTTER_ASSET_DIR_NAME}\")
  " COMPONENT Runtime)
install(DIRECTORY "${PROJECT_BUILD_DIR}/${FLUTTER_ASSET_DIR_NAME}"
  DESTINATION "${INSTALL_BUNDLE_DATA_DIR}" COMPONENT Runtime)

# Install the AOT library on non-Debug builds only.
if(NOT CMAKE_BUILD_TYPE MATCHES "Debug")
  install(FILES "${AOT_LIBRARY}" DESTINATION "${INSTALL_BUNDLE_LIB_DIR}"
    COMPONENT Runtime)
endif()
